cmake_minimum_required(VERSION 3.16)
project(robot_dog_firmware VERSION 0.1.0 LANGUAGES CXX)
option(ROBOT_ENABLE_SANITIZERS "Enable host ASan and UBSan" OFF)
option(ROBOT_BUILD_HOST_TESTS "Build host tests and dry-run only" ON)
add_library(robot_core STATIC src/lx16a.cpp src/bus.cpp src/safety.cpp src/kinematics.cpp)
target_include_directories(robot_core PUBLIC include)
target_compile_features(robot_core PUBLIC cxx_std_17)
set_target_properties(robot_core PROPERTIES CXX_EXTENSIONS OFF)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(robot_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(ROBOT_ENABLE_SANITIZERS)
    target_compile_options(robot_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(robot_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  endif()
elseif(ROBOT_ENABLE_SANITIZERS)
  message(FATAL_ERROR "This sanitizer configuration requires GCC or Clang")
endif()
if(ROBOT_BUILD_HOST_TESTS)
  enable_testing()
  foreach(name protocol bus safety kinematics)
    add_executable(test_${name} tests/test_${name}.cpp)
    target_include_directories(test_${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(test_${name} PRIVATE robot_core)
    add_test(NAME ${name} COMMAND test_${name})
  endforeach()
  add_executable(dry_run examples/dry_run.cpp)
  target_include_directories(dry_run PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_link_libraries(dry_run PRIVATE robot_core)
  add_test(NAME dry_run COMMAND dry_run)
endif()
